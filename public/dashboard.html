${rep.author}: ${rep.text}
`).join(''); return `
${c.author}: ${c.text}
${isMyComment ? `Edit` : ''}
${timeAgo(c.createdAt)} • Reply
${repliesHtml}
`; }).join(''); return `
🍌
${post.author}
${timeAgo(post.createdAt)} ${post.edited ? '(edited)' : ''}
${isMyPost ? `EditDelete` : ''}
${post.text ? `
${post.text}
` : ''} ${post.image ? `` : ''}
${makeBtn('banana', '🍌')} ${makeBtn('love', '❤️')} ${makeBtn('haha', '😂')} ${makeBtn('rocket', '🚀')}
${commentsHtml}
`; } async function react(id, type, btn) { const res = await fetch(`/api/posts/${id}/react`, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ type }) }); if (res.ok) { loadPosts(); // Reloads posts with updated names and correct counts } }
